function getSaleReferencePrefix(quote, documentKind) {
  const catalogKind = String(quote?.catalog_kind || "porton").toLowerCase().trim();
  const kind = String(documentKind || "pedido").toLowerCase().trim();
  if (catalogKind === "ipanel") return kind === "venta" ? "INV" : "INP";
  if (catalogKind === "puerta") return kind === "venta" ? "PNV" : "PNP";
  return kind === "venta" ? "NV" : "NP";
}
function buildSaleReference(quote, documentKind, number) {
  const n = toIntId(number);
  if (!n) return "";
  return getSaleReferencePrefix(quote, documentKind) + String(n);
}
async function ensureOrderReferenceNumber(quote) {
  const existing = toIntId(quote?.order_reference_number);
  if (existing) return existing;
  const id = toText(quote?.id);
  if (!id) throw new Error("Quote sin id para generar referencia NP/NV");

  const updated = await dbQuery(
    "update public.presupuestador_quotes\n"
      + "   set order_reference_number = nextval('public.presupuestador_order_reference_number_seq')\n"
      + " where id = $1\n"
      + "   and order_reference_number is null\n"
      + " returning order_reference_number",
    [id],
  );
  const next = toIntId(updated.rows?.[0]?.order_reference_number);
  if (next) {
    quote.order_reference_number = next;
    return next;
  }

  const current = await dbQuery(
    "select order_reference_number from public.presupuestador_quotes where id=$1 limit 1",
    [id],
  );
  const currentNumber = toIntId(current.rows?.[0]?.order_reference_number);
  if (!currentNumber) throw new Error("No se pudo generar número correlativo NP/NV");
  quote.order_reference_number = currentNumber;
  return currentNumber;
}
